cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(picostation_diag C CXX ASM)
pico_sdk_init()
add_executable(picostation_diag main.c)
target_compile_definitions(picostation_diag PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64)
target_link_libraries(picostation_diag pico_stdlib hardware_spi hardware_i2c)
pico_enable_stdio_usb(picostation_diag 1)
pico_enable_stdio_uart(picostation_diag 0)
pico_add_extra_outputs(picostation_diag)
